}

🔥 ✅ Readiness Example
Future<String> fetchData() async {
  await Future.delayed(Duration(seconds: 2));
  return "Data Loaded";
}

void main() async {
  print("Loading...");

  try {
    var data = await fetchData();
    print(data);
  } catch (e) {
    print("Error");
  }
}
